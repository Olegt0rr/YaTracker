94 """Decode one object out of a response documented as an array. 

95 

96 A handful of endpoints (create/edit a status, create/edit an

99 answer with a bare object. Both shapes are accepted here and the

101 

102 :raises ValueError: if the API answered with an empty array. 

103 """ 

104 decoded: Any = self._decode( 

105 list[type_] | type_, # type: ignore[valid-type,arg-type] 

106 data, 

107 ) 

108 if isinstance(decoded, list): 

109 if not decoded: 

110 msg = ( 

111 f"The API answered with an empty array where a single " 

112 f"{getattr(type_, '__name__', type_)} object was expected." 

113 ) 

114 raise ValueError(msg) 

115 decoded = decoded[0] 

116 return cast("T", decoded)

135 def _prepare_params(**kwargs: Any) -> dict[str, str] | None: # noqa: ANN401 

136 """Build query params from keyword arguments. 

137 

138 ``None`` values are dropped, booleans are encoded as ``"true"`` / 

139 ``"false"`` and everything else is stringified, because aiohttp 

140 (yarl) rejects ``bool`` and ``None`` query values with a

142 ``None`` when nothing is left, so the result can be passed to 

143 ``request(params=...)`` directly. 

144 """ 

145 params = { 

146 _encode_key(key): _encode_param(value) 

147 for key, value in kwargs.items() 

148 if value is not None

203def _check_sequence(values: object, param: str, item: str, example: str) -> list[Any]: 

204 """Materialize a collection passed where several items are expected. 

205 

206 Only the "bare single value" shapes are rejected: a `str`/`bytes` 

207 would be iterated character by character, a mapping key by key and a 

208 :class:`Base` model field by field, so the request would go out 

209 silently mangled instead of failing. Everything else iterable is 

210 accepted — a set, a `dict.keys()` view or a generator are legitimate 

211 ways to pass several ids — and returned as a list, so the callers 

212 can measure it and walk it more than once. 

213 

214 :param param: name of the offending parameter, as the caller wrote it. 

215 :param item: what the collection holds, e.g. "issue keys". 

216 :param example: one element, to show in the suggested call. 

217 :raises TypeError: if `values` is a single value rather than a 

218 collection of them. 

219 :return: the items as a list. 

220 """ 

221 if isinstance(values, (str, bytes, Mapping, Base)) or not isinstance( 

222 values, 

223 Iterable, 

224 ): 

225 msg = ( 

226 f"`{param}` must be a sequence of {item}, got " 

227 f"{type(values).__name__}. Pass a sequence, e.g. `[{example}]`." 

228 ) 

229 raise TypeError(msg) 

230 return list(values)

233def _join_fields(fields: str | Sequence[str] | None) -> str | None: 

234 """Render a `fields` projection as the comma-separated string the API wants. 

235 

236 Every endpoint taking `fields` documents it as one comma-separated 

237 string, which is also the most convenient shape when the projection 

238 is copied out of the reference. A sequence of names is joined here, 

239 so the same call works with `"key,summary"` and with 

240 `["key", "summary"]`. An empty value (`None`, `""`, `[]`) means 

241 "no projection" and is dropped by the callers. 

242 """ 

243 if not fields: 

244 return None 

245 return fields if isinstance(fields, str) else ",".join(fields)

248def _split_fields(fields: str | Sequence[str] | None) -> list[str] | None: 

249 """Render a `fields` projection as the JSON array some bodies want. 

250 

251 The mirror image of :func:`_join_fields`: a couple of endpoints 

252 (the report export, for one) take the projection as an array inside

275async def _iter_relative( 

276 fetch_page: Callable[[str | None], Awaitable[PageT]], 

277 *, 

278 items: Callable[[PageT], Sequence[T]], 

279 key: Callable[[T], str], 

280 has_next: Callable[[PageT], bool] | None = None, 

281) -> AsyncIterator[T]: 

282 """Walk over an id-cursor paginated endpoint, page by page. 

283 

284 Several Tracker endpoints (boards, triggers, users, changelog) share 

285 the same relative pagination: items are sorted by id ascending and 

286 the next page is requested with the id of the last item of the 

287 current one. The docs describe that `id` as the item the next page 

288 *starts from*, so the cursor item comes back at the top of the next 

289 page and is filtered out here instead of being yielded twice. 

290 

291 A page that does not advance past the cursor is either the last one 

292 (it only repeats the cursor item) or a server ignoring `id`: the 

293 iteration stops before yielding anything, even when the endpoint 

294 reports a next page, rather than looping forever. 

295 

296 :param fetch_page: coroutine fetching one page for a cursor 

297 (`None` for the first page). 

298 :param items: extract the items of a page. 

299 :param key: extract the cursor value of an item. 

300 :param has_next: whether the endpoint reports more pages; endpoints 

301 that do not report it stop on the first non-advancing page. 

302 """ 

303 cursor: str | None = None 

304 while True: 

305 page = await fetch_page(cursor) 

306 batch = items(page) 

307 if not batch or key(batch[-1]) == cursor: 

308 return 

309 

310 for item in batch: 

311 if key(item) != cursor: 

312 yield item 

313 

314 if has_next is not None and not has_next(page): 

315 return 

316 

317 cursor = key(batch[-1])

320def _relative_page_size(per_page: int | None) -> int | None: 

321 """Bump a one-item page size for an inclusive id cursor. 

322

324 item can only ever contain the cursor itself: the iteration would 

325 stop right after the first item. Two is the smallest size that still

338def _convert_value(obj: Any) -> Any: # noqa: ANN401 

339 """Convert values to basic types. 

340 

341 A model is rendered through its own request form (`Base._to_request`) 

342 rather than dumped verbatim, so a model read back from the API can 

343 be passed straight into the next request body. 

344 

345 Every collection is rendered as a JSON array, tuples and sets 

346 included: `pydantic_core.to_json` would serialize them, but without 

347 walking into them, so a model inside one would be dumped verbatim 

348 instead of going through its request form. A set is sorted when its 

349 elements are comparable, to keep the request reproducible, and left 

350 in iteration order otherwise. 

351 """ 

352 match obj: 

353 case Base(): 

354 return obj._to_request() # noqa: SLF001 

355 case list() | tuple(): 

356 return [_convert_value(o) for o in obj] 

357 case set() | frozenset(): 

358 return [_convert_value(o) for o in _sorted_if_possible(obj)] 

359 case dict(): 

360 return {k: _convert_value(v) for k, v in obj.items()} 

361 case _: 

362 return obj

373def _rename_and_clear( 

374 type_: type[Base] | None, 

375 payload: dict[str, Any], 

376 exclude: Collection[str], 

377) -> dict[str, Any]: 

378 """Replace kwarg keys with the model's encoded field names. 

379 

380 Keys that are not fields of `type_` (e.g. `query`, `filter_`, 

381 custom fields passed via **kwargs), or every key when `type_` is 

382 None, are kept and converted to camelCase instead of being dropped. 

383 Keys that are not Python identifiers (local-field ids like 

384 ``<id>--userId``) are kept as is. Two keys that would land on the 

385 same wire name raise `ValueError` rather than silently overwriting 

386 each other. 

387 """ 

388 renamed: dict[str, Any] = {} 

389 sources: dict[str, str] = {} 

390 exclude = {"self", "cls", *exclude} 

391 encode_names = _field_names(type_) if type_ is not None else {} 

392 

393 for name, raw_value in payload.items(): 

394 if name in exclude or name.startswith("_"): 

395 continue 

396 

397 value = _convert_value(raw_value) 

398 if value is None: 

399 continue 

400 

401 wire_name = encode_names.get(name) or _encode_key(name) 

402 if wire_name in renamed: 

403 msg = ( 

404 f"Payload keys {sources[wire_name]!r} and {name!r} both map " 

405 f"to the API field {wire_name!r}; pass only one of them." 

406 ) 

407 raise ValueError(msg) 

408 

409 renamed[wire_name] = value 

410 sources[wire_name] = name 

411 

412 return renamed
