12class EntityComment(Base): 

13 """Comment on an entity (a project, a portfolio or a goal). 

14 

15 Unlike an issue comment (`Comment`), an entity comment carries a 

16 string id next to the numeric one (`longId`), the rendered HTML of 

17 the text, the attachments and the reactions, so it is modelled 

18 separately. 

19 

20 The fields that depend on the `expand` query parameter are optional: 

21 `textHtml` needs "html", `attachments` needs "attachments" and 

22 `usersReacted`/`ownReactions` need "reactions" (or "all" for 

23 everything). Without "reactions" the API sends `reactionsCount` 

24 instead of `usersReacted`. 

25 

26 Source: 

27 https://yandex.ru/support/tracker/ru/api/entities/comments/get-comment 

28 """ 

29 

30 url: str = url_field() 

31 id: int 

32 long_id: str | None = None 

33 text: str 

34 text_html: str | None = None 

35 attachments: list[Ref] | None = None 

36 created_by: User 

37 updated_by: User | None = None 

38 created_at: datetime 

39 updated_at: datetime | None = None 

40 # `{"like": [<user>, ...]}`; the reaction names are server-owned 

41 # ("like", "dislike", "laugh", "tada", "hooray", "confused", "heart", 

42 # "rocket", "eyes", "fire", "ok", "facepalm", "check"). 

43 users_reacted: dict[str, list[User]] | None = None 

44 reactions_count: dict[str, int] | None = None 

45 own_reactions: list[str] | None = None 

46 # The samples show objects, but the docs describe both `summonees` 

47 # and `maillistSummonees` as an array of objects *or* of strings 

48 # (logins / ids), so plain strings are accepted as well. 

49 summonees: list[User | str] | None = None 

50 maillist_summonees: list[Ref | str] | None = None 

51 version: int 

52 #: "standard" (sent from the Tracker interface), "incoming" or 

53 #: "outcoming" (created from a mail). 

54 type: str | None = None 

55 #: "internal" (the Tracker interface) or "email".
